    camaras {
        int         id          PK
        varchar     nombre
        varchar     ubicacion
        text        rtsp_url
        boolean     activa
        datetime    created_at
    }

    sesiones_video {
        int         id          PK
        int         camara_id   FK
        datetime    inicio
        datetime    fin
        text        archivo_path
        float       duracion_seg
    }

    zonas {
        int         id          PK
        int         camara_id   FK
        varchar     nombre
        enum        tipo
        json        poligono
    }

    personas {
        int         id                          PK
        int         sesion_id                   FK
        datetime    primera_deteccion
        datetime    ultima_deteccion
        float       duracion_total_seg
        boolean     comportamiento_sospechoso
    }

    trayectorias {
        bigint      id          PK
        int         persona_id  FK
        int         zona_id     FK
        datetime    timestamp
        float       centroide_x
        float       centroide_y
        float       bbox_x1
        float       bbox_y1
        float       bbox_x2
        float       bbox_y2
    }

    productos {
        int         id          PK
        varchar     nombre
        varchar     codigo
        varchar     categoria
        int         stock_actual
        int         stock_minimo
        text        imagen_path
        boolean     activo
        datetime    created_at
    }

    detecciones_producto {
        bigint      id          PK
        int         sesion_id   FK
        int         producto_id FK
        datetime    timestamp
        float       confianza
        int         cantidad
        float       bbox_x1
        float       bbox_y1
        float       bbox_x2
        float       bbox_y2
    }

    alertas {
        int         id          PK
        int         persona_id  FK
        enum        tipo
        datetime    timestamp
        text        descripcion
        boolean     resuelta
        text        imagen_path
    }

    metricas_flujo {
        int         id                          PK
        int         camara_id                   FK
        int         zona_id                     FK
        datetime    periodo_inicio
        datetime    periodo_fin
        enum        granularidad
        int         personas_unicas
        int         personas_max_simultaneas
        float       permanencia_promedio_seg
        float       tiempo_espera_caja_seg
    }

    mapas_calor {
        int         id          PK
        int         camara_id   FK
        datetime    periodo_inicio
        datetime    periodo_fin
        enum        granularidad
        json        matriz
        int         resolucion_x
        int         resolucion_y
    }
